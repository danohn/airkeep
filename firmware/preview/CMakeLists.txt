cmake_minimum_required(VERSION 3.19)
project(airkeep_preview C CXX)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../managed_components/lvgl__lvgl lvgl)
add_executable(airkeep-preview preview.c ../main/ui.c ../main/font_digits.c)
target_include_directories(airkeep-preview PRIVATE stubs ../main)
target_link_libraries(airkeep-preview PRIVATE lvgl)
enable_testing()
add_test(NAME footer-touch COMMAND airkeep-preview --test-touch)
add_test(NAME relay-ui-state COMMAND airkeep-preview --test-relay-state)

# Reuse cJSON from the installed SDK without downloading another copy.
set(AIRKEEP_CJSON_DIR "$ENV{IDF_PATH}/components/json/cJSON" CACHE PATH "Existing ESP-IDF cJSON source")
if(NOT EXISTS "${AIRKEEP_CJSON_DIR}/cJSON.c" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../build/project_description.json")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../build/project_description.json" firmware_project)
    string(JSON installed_idf GET "${firmware_project}" idf_path)
    set(AIRKEEP_CJSON_DIR "${installed_idf}/components/json/cJSON")
endif()
if(NOT EXISTS "${AIRKEEP_CJSON_DIR}/cJSON.c")
    message(FATAL_ERROR "Set AIRKEEP_CJSON_DIR to cJSON inside an existing ESP-IDF installation")
endif()
add_executable(airkeep-snapshot-test ../tests/relay_snapshot_test.c ../main/relay_snapshot.c "${AIRKEEP_CJSON_DIR}/cJSON.c")
target_include_directories(airkeep-snapshot-test PRIVATE ../main "${AIRKEEP_CJSON_DIR}")
if(UNIX)
    target_link_libraries(airkeep-snapshot-test PRIVATE m)
endif()
add_test(NAME relay-snapshot COMMAND airkeep-snapshot-test)
